\section{Personnes conseillères nationales}
\label{personnes-conseilleres-nationales}

\subsection{Objectif}
\label{personnes-conseilleres-nationales-objectif}

\begin{enumerate}
 \item Les personnes conseillères nationales fourniront au CA une continuité et des connaissances institutionnelles, et donneront des conseils sur toute question qui leur sera soumise.
 \item Les personnes conseillères nationales seront responsables d'aider le processus de transition organisationnelle.
\end{enumerate}

\subsection{Élection}
\label{personnes-conseilleres-nationales-election}

\begin{enumerate}
 \item Les deux (2) personnes conseillères nationales seront élues lors de l'assemblée générale annuelle.
 \item Il n'y a aucune condition d'éligibilité spécifique pour la candidature à un poste de personne conseillère nationale.
\end{enumerate}

\subsection{Mandat}
\label{personnes-conseilleres-nationales-mandat}

\begin{enumerate}
 \item Le mandat des personnes conseillères nationales commencera à la date de leur nomination et expirera à la clôture de l'assemblée générale annuelle de l'année suivante.
\end{enumerate}
