\section{Conseil d'administration}
\label{conseil-d-administration}

\subsection{Objectif}
\label{conseil-d-administration-objectif}

\begin{enumerate}
 \item Le CA exercera une surveillance fiduciaire et de gouvernance de la FCEG, et gérera ses biens et ses affaires pour l'avancement de sa mission.
\end{enumerate}

\subsection{Composition}
\label{conseil-d-administration-composition}

\begin{enumerate}
 \item Il y aura huit (8) personnes administratrices:
  \begin{enumerate}
   \item La présidence ;
   \item La présidence du CA ;
   \item Les quatre (4) personnes aux ambassades régionales ;
   \item Les deux (2) personnes conseillères nationales ;
  \end{enumerate}
\end{enumerate}

\subsection{Pouvoirs d'administration}
\label{conseil-d-administration-pouvoirs-d-administration}

\begin{enumerate}
 \item Le CA peut administrer les affaires de la FCEG en toutes choses et conclure ou faire conclure pour la FCEG, en son nom, tout type de contrat que la FCEG peut légalement conclure et, sauf disposition contraire ci-après, peut généralement exercer tous les autres pouvoirs et accomplir tous les autres actes et affaires que la FCEG est autorisée à exercer et à accomplir par ses règlements ou autrement.
\end{enumerate}

\subsection{Vote}
\label{conseil-d-administration-vote}

\begin{enumerate}
 \item En cas d'égalité des voix, la motion sera rejetée.
\end{enumerate}
