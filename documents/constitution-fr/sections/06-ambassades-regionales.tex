\section{Ambassades régionales}
\label{ambassades-regionales}

\subsection{Nomination}
\label{ambassades-regionales-nomination}

\begin{enumerate}
 \item Chaque région devra, avant le SCIP de chaque année, communiquer à la présidence du CA sa personne candidate pour le poste d'ambassade régionale de sa région.
 \item La nomination des personnes aux ambassades régionales sera approuvée par une résolution d'une assemblée générale.
\end{enumerate}

\subsection{Éligibilité}
\label{ambassades-regionales-eligibilite}

\begin{enumerate}
 \item La personne candidate doit faire partie d'une association membre de cette région au moment de sa ratification.
\end{enumerate}

\subsection{Mandat}
\label{ambassades-regionales-mandat}

\begin{enumerate}
 \item Le mandat des personnes aux ambassades régionales commencera à la date de leur nomination et expirera à la clôture de l'assemblée générale annuelle de l'année suivante.
\end{enumerate}
