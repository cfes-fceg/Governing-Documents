\section{Assemblées générales}
\label{assemblees-generales}

\subsection{Assemblée générale annuelle}
\label{assemblees-generales-assemblee-generale-annuelle}

\begin{enumerate}
 \item L'assemblée générale annuelle se tiendra au SCIP.
 \item Les points à l'ordre du jour traités lors de l'assemblée générale annuelle comprendront :
  \begin{enumerate}
   \item La réception de l'ordre du jour,
   \item La réception du procès-verbal de l'assemblée générale précédente,
   \item La nomination de l’auditeur pour l'année suivante,
   \item La présentation des états financiers audités de l'exercice précédent, ainsi que le rapport de l’auditeur,
   \item L'élection de l'équipe de direction,
   \item La présentation de toute autre information ou rapport relatif aux activités et aux affaires de la FCEG que le CA peut déterminer.
  \end{enumerate}
\end{enumerate}

\subsection{Assemblées générales}
\label{assemblees-generales-assemblees-generales}

\begin{enumerate}
 \item Les assemblées générales se tiendront au SDAI, au CCLI et au SCIP.
\end{enumerate}

\subsection{Assemblées extraordinaires des membres}
\label{assemblees-generales-assemblees-extraordinaires-des-membres}

\begin{enumerate}
 \item Une assemblée extraordinaire des membres peut être convoquée par une résolution du CA, ou par une pétition signée par 5 \% des membres et dûment soumise à la présidence du CA.
 \item La présidence du CA convoquera l'assemblée extraordinaire dans les vingt-et-un (21) jours suivant la réception de la pétition ou de la résolution. Si la présidence du CA ne convoque pas l'assemblée extraordinaire dans les vingt-et-un (21) jours suivant la réception de la pétition, tout membre ayant signé la pétition peut convoquer l'assemblée.
\end{enumerate}
