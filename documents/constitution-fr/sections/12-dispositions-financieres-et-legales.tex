\section{Dispositions financières et légales}
\label{dispositions-financieres-et-legales}

\subsection{Audit externe}
\label{dispositions-financieres-et-legales-audit-externe}

\begin{enumerate}
 \item La FCEG doit effectuer un audit externe chaque année.
\end{enumerate}

\subsection{Audit par les membres}
\label{dispositions-financieres-et-legales-audit-par-les-membres}

\begin{enumerate}
 \item Lors de toute assemblée générale, un membre peut demander à consulter les livres comptables.
\end{enumerate}

\subsection{Remboursements}
\label{dispositions-financieres-et-legales-remboursements}

\begin{enumerate}
 \item Le CA peut autoriser le remboursement à l'équipe de la FCEG de tous les coûts, frais et dépenses qu'ils subissent ou engagent dans ou à propos ou en relation avec les affaires de celle-ci, à l'exception des coûts, frais ou dépenses occasionnés par leur propre négligence ou manquement intentionnel.
 \item L'équipe de direction sera remboursée de tous les coûts, frais et dépenses subis ou engagés à propos de toute action, poursuite ou procédure intentée, commencée ou poursuivie contre eux, ou relativement à tout acte, document, affaire ou chose quelconque, fait, accompli ou permis par eux, dans ou à propos de l'exécution des devoirs de leur poste ou relativement à une telle responsabilité.
\end{enumerate}

\subsection{Société de fiducie}
\label{dispositions-financieres-et-legales-societe-de-fiducie}

\begin{enumerate}
 \item Le CA aura le pouvoir d'autoriser des dépenses au nom de la FCEG de temps à autre et pourra déléguer par résolution à un membre de l'équipe de direction le droit d'employer et de payer des salaires au personnel.
 \item Le CA aura le pouvoir de conclure un accord de fiducie avec une société de fiducie dans le but de créer un fonds en fiducie dont le capital et les intérêts pourront être mis à disposition pour promouvoir les intérêts de la FCEG conformément aux conditions que le CA ou les membres pourront prescrire.
\end{enumerate}

\subsection{Dons}
\label{dispositions-financieres-et-legales-dons}

\begin{enumerate}
 \item Le CA prendra les mesures qu'il jugera nécessaires pour permettre à la FCEG d'acquérir, d'accepter, de solliciter ou de recevoir des legs, des cadeaux, des subventions, des règlements, des fondations et des dons de toute nature que ce soit dans le but de promouvoir les objectifs de la FCEG.
\end{enumerate}

\subsection{Procuration}
\label{dispositions-financieres-et-legales-procuration}

\begin{enumerate}
 \item Le CA peut donner procuration à tout courtier en valeurs mobilières inscrit pour le transfert et la négociation de toutes actions, obligations et autres valeurs mobilières de la FCEG.
\end{enumerate}
