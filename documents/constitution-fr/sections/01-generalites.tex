\section{Généralités}
\label{generalites}

\subsection{Nom de l'organisation}
\label{generalites-nom-de-l-organisation}

\begin{enumerate}
 \item Le nom de l'organisation est « La Fédération canadienne des étudiantes en génie ».
\end{enumerate}

\subsection{Exercice financier}
\label{generalites-exercice-financier}

\begin{enumerate}
 \item L'exercice financier de la FCEG commence le 1er mai et se termine le 30 avril de l'année suivante.
\end{enumerate}

\subsection{Logo}
\label{generalites-logo}

\begin{enumerate}
 \item Le logo qui apparaît ci-dessous est utilisé pour désigner la Fédération canadienne étudiante en génie.
\end{enumerate}
