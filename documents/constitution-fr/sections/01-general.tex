\section{Article 1 - Général}\label{article-1---guxe9nuxe9ral}

\subsection{Le nom de la corporation}\label{le-nom-de-la-corporation}

\begin{enumerate}
 \item
  Le nom de la corporation sera « la Fédération Canadienne Étudiante en Génie », désignée ci-après par « la Fédération ».
 \item
  Le nom anglais officiel de la Fédération sera « Canadian Federation of Engineering Students ».
\end{enumerate}

\subsection{Le siège social}\label{le-siuxe8ge-social}

\begin{enumerate}
 \item
  Le siège social de la Fédération sera situé au bureau d'Ingénieurs Canada, 55 rue Metcalfe - Bureau 300, Ottawa, Ontario, K1P 6L5.
\end{enumerate}

\subsection{Le sceau}\label{le-sceau}

\begin{enumerate}
 \item
  Le sceau, dont une impression est apposée ci-dessous, sera le sceau de la Fédération.
\end{enumerate}

\subsection{L'année fiscale}\label{lannuxe9e-fiscale}

\begin{enumerate}
 \item
  L'année fiscale de la Fédération se terminera le 30 avril de chaque année.
\end{enumerate}

\subsection{L'abbréviation}\label{labbréviation}

\begin{enumerate}
 \item
  Les termes « FCEG » et « CFES » sont employés pour désigner la Fédération Canadienne Étudiante en Génie.
\end{enumerate}
\subsection{Le logo}\label{le-logo}
\begin{enumerate}
 \item
  Le logo ci-dessous est employé pour désigner la Fédération Canadienne Étudiante en Génie.
\end{enumerate}
