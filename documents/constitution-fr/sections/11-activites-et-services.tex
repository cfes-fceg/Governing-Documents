\section{Activités et services}
\label{activites-et-services}

\subsection{Aperçu}
\label{activites-et-services-apercu}

\begin{enumerate}
 \item Les activités de la FCEG incluront le CCLI, le CDDI, la CCI, le SDAI et le CDI.
\end{enumerate}

\subsection{Organisation des activités et des services}
\label{activites-et-services-organisation-des-activites-et-des-services}

\subsubsection{Associations hôtes}
\label{organisation-des-activites-et-des-services-associations-hotes}

\begin{enumerate}
 \item Les associations hôtes seront nommées par une résolution de l'assemblée générale.
\end{enumerate}

\subsubsection{Personnes responsables d'activités}
\label{organisation-des-activites-et-des-services-personnes-responsables-d-activites}

\begin{enumerate}
 \item Les personnes responsables d'activités seront nommées par l'association hôte.
 \item Leur mandat s'étendra du moment de la nomination jusqu'à la résolution de tous les comptes clients ou fournisseurs à la conclusion de l'activité en question.
\end{enumerate}
