\section{Élections partielles}
\label{elections-partielles}

\begin{enumerate}
 \item Les postes vacants peuvent être pourvus par une élection partielle.
 \item Les procédures d'élection partielle seront établies par la DGE.
 \item Les résultats des élections partielles seront ratifiés par résolution de l'assemblée générale suivante.
\end{enumerate}
