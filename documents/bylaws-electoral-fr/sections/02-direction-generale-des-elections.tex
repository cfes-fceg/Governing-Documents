\section{Direction générale des élections}
\label{direction-generale-des-elections}

\begin{enumerate}
 \item La DGE sera responsable de la conduite de toute élection et mise en candidature de la FCEG.
 \item La présidence du CA agira à titre de DGE d'office.
  \begin{enumerate}
   \item Le CA peut, par résolution spéciale, nommer une autre personne pour agir à titre de DGE pour toute élection.
   \item En cas de conflit d'intérêt, une telle nomination sera obligatoire.
  \end{enumerate}
 \item La DGE sera responsable de la résolution de tout conflit, ambiguïté et interprétation des procédures électorales et de mise en candidature.
 \item Toutes les décisions de la DGE seront finales.
\end{enumerate}
