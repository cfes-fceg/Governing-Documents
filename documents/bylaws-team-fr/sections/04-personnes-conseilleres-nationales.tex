\section{Personnes conseillères nationales}
\label{personnes-conseilleres-nationales}

\subsection{Objectif}
\label{personnes-conseilleres-nationales-objectif}

\begin{enumerate}
 \item La FCEG élira deux (2) personnes conseillères nationales chaque année pour siéger à titre de personnes administratrices et soutenir sa mission.
\end{enumerate}

\subsection{Responsabilités}
\label{personnes-conseilleres-nationales-responsabilites}

\begin{enumerate}
 \item Les responsabilités des personnes conseillères nationales incluent :
  \begin{enumerate}
   \item Conseiller et soutenir l'équipe de la FCEG et les membres avec des connaissances institutionnelles
   \item Assurer une transition fluide de l'équipe de la FCEG
  \end{enumerate}
\end{enumerate}
