\section{Chair}
\label{chair}

\subsection{Objectif}
\label{chair-objectif}

\begin{enumerate}
 \item Assurer le leadership et la supervision du CA et des assemblées générales.
 \item Faciliter une gouvernance et une prise de décision ordonnées, transparentes et efficaces.
 \item S'assurer que les réunions sont correctement convoquées, préparées, menées et documentées conformément à la constitution et aux règlements.
 \item Soutenir une communication et une participation efficaces parmi les membres et l'équipe de direction.
 \item S'assurer que l'organisation maintient une gouvernance d'entreprise adéquate et remplit ses obligations en vertu de la LCOBNL et des lois applicables.
 \item Agir en tant que principal point de responsabilité de l'organisation pour l'incorporation et les questions de conformité d'entreprise en cours.
\end{enumerate}

\subsection{Responsabilités}
\label{chair-responsabilites}

\begin{enumerate}
 \item Présider les réunions du CA, y compris :
  \begin{enumerate}
   \item Convoquer les réunions du CA conformément à la constitution.
   \item Distribuer les procès-verbaux des réunions au CA pour approbation.
   \item Soumettre les procès-verbaux approuvés pour distribution publique.
  \end{enumerate}
 \item Présider les assemblées générales, y compris :
  \begin{enumerate}
   \item S'assurer, conjointement avec la personne responsable de l'activité concernée, que suffisamment de temps et de ressources sont prévus pour mener à bien toutes les procédures,
   \item S'assurer que les membres sont informés des réunions tel que requis par la constitution,
   \item S'assurer que les membres sont adéquatement préparés pour la réunion en leur fournissant les procès-verbaux des réunions précédentes et tout autre document pertinent,
   \item S'assurer qu'un processus existe pour que les membres puissent soumettre des propositions et des contre-propositions,
   \item S'assurer qu'un ordre du jour préliminaire est diffusé tel que requis pour chaque réunion.
  \end{enumerate}
 \item En l'absence de la présidence du CA, les responsabilités ci-dessus doivent être complétées par la vice-présidence du CA.
  \begin{enumerate}
   \item Le CA désignera un membre de l'équipe de direction à titre de vice-présidence du CA, qui remplira les fonctions de la présidence du CA en son absence.
   \item La présidence ne sera pas éligible pour pourvoir le poste de vice-présidence du CA.
  \end{enumerate}
 \item Agir à titre de direction générale des élections d'office
 \item Traiter toutes les questions d'incorporation, y compris le dépôt du résumé annuel auprès de Corporations Canada chaque année.
 \item Assurer la conformité continue avec la LCOBNL et les lois applicables.
\end{enumerate}
