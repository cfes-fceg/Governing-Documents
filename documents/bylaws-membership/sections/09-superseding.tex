\section{Superseding}
\label{superseding}

\begin{enumerate}
 \item In the event of a conflict between these Bylaws and the following documents, the following documents shall prevail in the order in which they are listed:
  \begin{enumerate}
   \item The Canada Not-For-Profit Corporations Act
   \item The Articles
   \item The Constitution
  \end{enumerate}
\end{enumerate}
