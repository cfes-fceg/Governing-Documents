\section{General Meetings}
\label{general-meetings}

\subsection{Meeting Guidelines}
\label{general-meetings-meeting-guidelines}

\subsubsection{Calling a Meeting}
\label{meeting-guidelines-calling-a-meeting}

\begin{enumerate}
 \item A General Meeting shall be called upon Resolution of the Board or on a written requisition of no less than 5\% of the Members.
\end{enumerate}

\subsubsection{Live Translation}
\label{meeting-guidelines-live-translation}

\begin{enumerate}
 \item At every General Meeting, simultaneous translation services shall be available, whether they be professionally outsourced or member-led.
\end{enumerate}

\subsubsection{Notice of a meeting}
\label{meeting-guidelines-notice-of-a-meeting}

\begin{enumerate}
 \item A twenty-one (21) day written notice shall be given to Members of any General Meeting.
 \item No error or omission in giving notice of any General Meeting of the CFES shall invalidate such meeting or make void any proceedings taken thereat.
\end{enumerate}

\subsubsection{Voting and Quorum}
\label{meeting-guidelines-voting-and-quorum}

\begin{enumerate}
 \item All Members shall have one (1) vote.
 \item All questions before the General Meeting shall be determined by a majority of votes of Members present and voting unless otherwise specified.
 \item Two thirds (2/3) of the Members shall constitute quorum at a General Meeting.
\end{enumerate}

\subsubsection{Speaking Rights}
\label{meeting-guidelines-speaking-rights}

\begin{enumerate}
 \item Members and the Leadership Team shall have speaking rights during General Meetings.
\end{enumerate}

\subsubsection{Order}
\label{meeting-guidelines-order}

\begin{enumerate}
 \item At all General Meetings, the Chair must apply the rules of deliberating assemblies as outlined in the Robert’s Rules of Order, except where contrary dispositions are adopted by the CFES.
 \item The Chair shall preside over General Meetings.
\end{enumerate}

\subsubsection{Representative}
\label{meeting-guidelines-representative}

\begin{enumerate}
 \item A Member shall, by means of verifiable communication by its President, appoint a representative to attend and act at a specific General Meeting.
 \item Notice of each General Meeting must remind the Members of their ability to appoint a representative.
 \item All representatives must be a member student of the Member Society or of another Member Society of the CFES.
 \item All proxy voting regulations shall additionally comply with Section 74 of the Canada Not-for-Profit Corporations Regulations.
\end{enumerate}

\subsubsection{Minutes}
\label{meeting-guidelines-minutes}

\begin{enumerate}
 \item Minutes shall be taken in both English and French at all General Meetings.
 \item The full and proper text of the minutes shall be distributed to all Members in both English and French within thirty days of the adjournment of the meeting.
\end{enumerate}

\subsection{Pre-General Meeting Videos}
\label{general-meetings-pre-general-meeting-videos}

\begin{enumerate}
 \item The National Executive may create videos prior to any Activity that involves a General Meeting to ensure members are informed about relevant topics to the conference; including but not limited to the policies being discussed and ongoing mandates.
 \item The total duration of these videos shall not exceed 30 minutes, and videos must be provided a minimum of 1 week before the conference begins.
\end{enumerate}
