\section{Chair}
\label{chair}

\subsection{Purpose}
\label{chair-purpose}

\begin{enumerate}
 \item The Chair shall be responsible for providing leadership to the Board and ensuring the effective governance of the CFES.
 \item The Chair shall facilitate the orderly conduct of meetings of the Board and the Members, support effective decision-making, and promote continuity and accountability within the CFES.
 \item The Chair shall serve as the presiding officer of the Board and the General Meetings.
\end{enumerate}

\subsection{Term}
\label{chair-term}

\begin{enumerate}
 \item The term of office for the Chair shall commence on the date of their appointment and expire at the close of the next year’s Annual General Meeting.
\end{enumerate}

\subsection{Appointment}
\label{chair-appointment}

\begin{enumerate}
 \item The Board shall appoint a Chair from among members of Member Societies. The appointment of the Chair shall constitute the appointment of an additional Director.
 \item The appointment of the Chair shall be submitted to the Members for ratification at the Annual General Meeting.
\end{enumerate}
