\section{Board of Directors}
\label{board-of-directors}

\subsection{Purpose}
\label{board-of-directors-purpose}

\begin{enumerate}
 \item The Board shall exercise fiduciary and governance oversight of the CFES, and shall manage its property and business in furtherance of its mission.
\end{enumerate}

\subsection{Composition}
\label{board-of-directors-composition}

\begin{enumerate}
 \item There shall be eight (8) Directors
  \begin{enumerate}
   \item The President;
   \item The Chair;
   \item The four (4) Regional Ambassadors;
   \item The two (2) National Councillors;
  \end{enumerate}
\end{enumerate}

\subsection{Powers of Administration}
\label{board-of-directors-powers-of-administration}

\begin{enumerate}
 \item The Board may administer the affairs of the CFES in all things and make or cause to be made for the CFES, in its name, any kind of contract which the CFES may lawfully enter into and, save as hereinafter provided, generally, may exercise all such other powers and do all such other acts and matters of the CFES is by its by-laws or otherwise authorized to exercise and do.
\end{enumerate}

\subsection{Voting}
\label{board-of-directors-voting}

\begin{enumerate}
 \item In the case of an equality of votes, the motion shall be defeated.
\end{enumerate}
