\section{National Executive}
\label{national-executive}

\subsection{Composition}
\label{national-executive-composition}

\begin{enumerate}
 \item The National Executive shall consist of the:
  \begin{enumerate}
   \item President
   \item Vice President Academic
   \item Vice President Communications
   \item Vice President External
   \item Vice President Finance
   \item Vice President Services
  \end{enumerate}
\end{enumerate}

\subsection{Election}
\label{national-executive-election}

\begin{enumerate}
 \item The National Executive shall be elected at the Annual General Meeting.
\end{enumerate}

\subsection{Eligibility}
\label{national-executive-eligibility}

\begin{enumerate}
 \item The candidate must be a member of a Member Society at the time of their candidature.
\end{enumerate}

\subsection{Term of Office}
\label{national-executive-term-of-office}

\begin{enumerate}
 \item The term of office for the National Executive shall commence on the date of their appointment and expire at the close of the next year’s Annual General Meeting.
 \item Concluding their term in office, the Vice-President Finance will transition to the role of Outgoing Vice-President Finance, with a term concluding on August 31st.
\end{enumerate}
