\section{Regional Ambassadors}
\label{regional-ambassadors}

\subsection{Appointment}
\label{regional-ambassadors-appointment}

\begin{enumerate}
 \item Each region shall, before CESS each year, communicate to the Chair their nominee for the position of Regional Ambassador to their region.
 \item The appointment of the Regional Ambassadors shall be approved by a Resolution of a General Meeting.
\end{enumerate}

\subsection{Eligibility}
\label{regional-ambassadors-eligibility}

\begin{enumerate}
 \item The candidate must be a member of a Member Society of that region at the time of their ratification.
\end{enumerate}

\subsection{Term}
\label{regional-ambassadors-term}

\begin{enumerate}
 \item The term of office for the Regional Ambassadors shall commence on the date of their appointment and expire at the close of the next year’s Annual General Meeting.
\end{enumerate}
