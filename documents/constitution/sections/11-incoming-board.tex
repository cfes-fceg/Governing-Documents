\section{Article 11 - Incoming Board}
\label{article-11---incoming-board}

\subsection{Purpose}
\label{purpose-incoming-board}

\begin{enumerate}
 \item The Incoming Board shall serve to facilitate the organizational transition process leading up to the commencement of their term of office.
\end{enumerate}

\subsection{Composition}
\label{composition-incoming-board}

\begin{enumerate}
 \item The Incoming Board shall consist of the following, as have been ratified by the General Assembly:
  \begin{enumerate}
   \item Incoming National Executive;
   \item Incoming Regional Ambassadors;
   \item Incoming National Councillors.
  \end{enumerate}
\end{enumerate}

\subsection{Term of Office}
\label{term-of-office-incoming-board}

\begin{enumerate}
 \item The term of office of the Incoming Board shall be from the date of ratification by the General Assembly until the commencement of the Fiscal Year on May 1st.
\end{enumerate}
