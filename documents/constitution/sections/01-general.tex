\section{General}
\label{general}

\subsection{Name of the Corporation}
\label{general-name-of-the-corporation}

\begin{enumerate}
 \item The name of the Corporation shall be “The Canadian Federation of Engineering Students”.
\end{enumerate}

\subsection{Fiscal Year}
\label{general-fiscal-year}

\begin{enumerate}
 \item The Fiscal Year of the CFES shall begin on May 1st and conclude on April 30th of the following year.
\end{enumerate}

\subsection{Logo}
\label{general-logo}

\begin{enumerate}
 \item The Logo that appears below is employed to designate the Canadian Federation of Engineering Students.
\end{enumerate}
