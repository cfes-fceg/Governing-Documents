\section{General Meetings}
\label{general-meetings}

\subsection{Annual General Meeting}
\label{general-meetings-annual-general-meeting}

\begin{enumerate}
 \item The Annual General Meeting shall be held at CESS.
 \item The business items transacted at the Annual General Meeting shall include:
  \begin{enumerate}
   \item Receipt of the agenda,
   \item Receipt of the minutes of the previous General Meeting,
   \item Appointment of the auditor for the ensuing year,
   \item Presentation of the audited financial statements of the previous financial year, together with the auditor's report,
   \item Election of the Leadership Team,
   \item Presentation of any other information or reports relating to the CFES's activities and affairs as the Board may determine.
  \end{enumerate}
\end{enumerate}

\subsection{General Meetings}
\label{general-meetings-general-meetings}

\begin{enumerate}
 \item General Meetings shall be held at SDES, CELC, and CESS.
\end{enumerate}

\subsection{Special Meetings of the Members}
\label{general-meetings-special-meetings-of-the-members}

\begin{enumerate}
 \item A Special Meeting of the Members may be called by a Resolution of the Board, or by a petition signed by 5\% of the Members duly submitted to the Chair.
 \item The Chair shall call the Special Meeting within twenty-one (21) days of the receipt of the petition or the Resolution. If the Chair does not call the Special Meeting within twenty-one (21) days of receiving the petition, any Member who signed the petition may call the meeting.
\end{enumerate}
