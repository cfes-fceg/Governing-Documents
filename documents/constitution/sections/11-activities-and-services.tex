\section{Activities and Services}
\label{activities-and-services}

\subsection{Overview}
\label{activities-and-services-overview}

\begin{enumerate}
 \item The Activities of the CFES shall include CELC, CSE, CEC, SDES, and CDE.
\end{enumerate}

\subsection{Organization of Activities and Services}
\label{activities-and-services-organization-of-activities-and-services}

\subsubsection{Host Societies}
\label{organization-of-activities-and-services-host-societies}

\begin{enumerate}
 \item Host Societies shall be appointed by a Resolution of the General Meeting.
\end{enumerate}

\subsubsection{Activity Managers}
\label{organization-of-activities-and-services-activity-managers}

\begin{enumerate}
 \item Activity Managers shall be appointed by the Host Society.
 \item Their term of office shall span from the time of appointment until the resolution of any accounts receivable or payable at the conclusion of the activity in question.
\end{enumerate}
