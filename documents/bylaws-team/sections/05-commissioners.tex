\section{Commissioners}
\label{commissioners}

\subsection{Nomination and Ratification}
\label{commissioners-nomination-and-ratification}

\begin{enumerate}
 \item The Board, on recommendation of the National Executive, may appoint the Commissioners for the purpose of assisting with the National Executives’ respective portfolios.
\end{enumerate}

\subsection{Term of Office}
\label{commissioners-term-of-office}

\begin{enumerate}
 \item The term of office of the Commissioners is the same as that of the National Executive.
\end{enumerate}

\subsection{Dismissal}
\label{commissioners-dismissal}

\begin{enumerate}
 \item A Commissioner may be dismissed by a Special Resolution of the Board..
\end{enumerate}

\subsection{Vacancy}
\label{commissioners-vacancy}

\begin{enumerate}
 \item Provided a vacancy occurs, the Board, on recommendation of the National Executive, may appoint a Commissioner by interim until the end of the term of office.
\end{enumerate}

\subsection{Remuneration}
\label{commissioners-remuneration}

\begin{enumerate}
 \item The Commissioners shall serve without remuneration and no Commissioner shall directly or indirectly receive any profit from holding the position, provided that reasonable expenses incurred in the exercise of the Commissioner’s duties are reimbursed.
\end{enumerate}
