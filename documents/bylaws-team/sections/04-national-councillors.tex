\section{National Councillors}
\label{national-councillors}

\subsection{Purpose}
\label{national-councillors-purpose}

\begin{enumerate}
 \item The CFES shall elect two (2) National Councillors annually to serve as Directors and to support its Mission.
\end{enumerate}

\subsection{Responsibilities}
\label{national-councillors-responsibilities}

\begin{enumerate}
 \item Responsibilities of the National Councillors include:
  \begin{enumerate}
   \item Advising and supporting the CFES Team and Members with institutional knowledge
   \item Ensuring a smooth transition of the CFES Team
  \end{enumerate}
\end{enumerate}
