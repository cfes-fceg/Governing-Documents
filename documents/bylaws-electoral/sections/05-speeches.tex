\section{Speeches}
\label{speeches}

\begin{enumerate}
 \item Candidates shall be allowed to present a speech to the Members of a length and order determined by the CRO.
 \item A question period shall be held after all candidates have spoken and shall be mediated by the CRO.
  \begin{enumerate}
   \item Each candidate shall be given the opportunity to answer any and all questions.
  \end{enumerate}
 \item Translation services shall be provided during the speeches and question period.
\end{enumerate}
