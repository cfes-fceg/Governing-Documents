\section{Autres réunions des membres}
\label{autres-reunions-des-membres}

\subsection{Camp de formation}
\label{autres-reunions-des-membres-camp-de-formation}

\begin{enumerate}
 \item Le camp de formation sera un rassemblement en ligne des membres et de l'équipe de la FCEG dont le but est d'accueillir des séances de formation et des discussions, ainsi que du réseautage.
 \item Le camp de formation aura lieu en juin ou en juillet.
\end{enumerate}

\subsection{Journées d’actualités saisonnières en téléconférence (JASET)}
\label{autres-reunions-des-membres-journees-d-actualites-saisonnieres-en-teleconference-jaset}

\begin{enumerate}
 \item La FCEG tiendra deux réunions à distance pour combler l'écart entre les assemblées générales.
 \item Ces réunions seront un rassemblement en ligne des membres et de l'équipe de la FCEG.
 \item L'objectif de ces réunions sera pour l'équipe de la FCEG de fournir des mises à jour aux membres sur les progrès de leurs portefeuilles, projets et travaux de revendication.
 \item Ces réunions comprendront également des opportunités pour les membres de fournir des commentaires.
 \item Ces réunions seront planifiées de manière à ce qu'une ait lieu avant le SDAI et une autre avant le CCLI.
 \item L'organisation de ces réunions et de leurs ordres du jour relève de la responsabilité de la présidence du CA avec le soutien des membres pertinents de l'équipe de la FCEG.
\end{enumerate}
