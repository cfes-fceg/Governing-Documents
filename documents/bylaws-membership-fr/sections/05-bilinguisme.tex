\section{Bilinguisme}
\label{bilinguisme}

\begin{enumerate}
 \item Les membres auront le droit de communiquer avec et de recevoir des communications de l'équipe de la FCEG dans la langue officielle de leur préférence.
 \item Les membres auront le droit de demander la traduction de tout document de la FCEG d'une langue officielle à l'autre.
\end{enumerate}
